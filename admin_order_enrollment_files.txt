const User = require("../models/User");
const Course = require("../models/Course");
const AdminStats = require("../models/AdminStats");

const adminService = {
  async getStats() {
    const [
      revenue, revenueByDay, activeLearners, newSignups,
      liveCourses, totalCourses, enrollments,
      topCourses, recentEnrollments,
    ] = await Promise.all([
      AdminStats.revenueTotals(),
      AdminStats.revenueByDay(7),
      User.countActive(),
      User.countNewByDay(30),
      Course.countLive(),
      Course.countAll(),
      AdminStats.totalEnrollments(),
      Course.topByEnrollments(5),
      AdminStats.recentEnrollments(5),
    ]);

    // Fill 7-day gaps so the chart always has 7 bars
    const revFilled = fillDays(revenueByDay, 7, { amount: 0 });
    const signFilled = fillDays(newSignups, 30, { count: 0 });

    return {
      totals: {
        revenue: revenue.revenue,
        paidOrders: revenue.paidOrders,
        activeLearners,
        liveCourses,
        totalCourses,
        totalEnrollments: enrollments,
      },
      revenueLast7Days: revFilled,
      signupsLast30Days: signFilled,
      topCourses,
      recentEnrollments,
    };
  },
};

function fillDays(rows, days, zeroShape) {
  const map = new Map(rows.map((r) => [r.date, r]));
  const out = [];
  const today = new Date();
  for (let i = days - 1; i >= 0; i--) {
    const d = new Date(today);
    d.setDate(today.getDate() - i);
    const key = d.toISOString().slice(0, 10);
    out.push(map.get(key) || { date: key, ...zeroShape });
  }
  return out;
}

module.exports = adminService;

const asyncHandler = require("../utils/asyncHandler");
const ApiResponse = require("../utils/ApiResponse");
const ApiError = require("../utils/ApiError");
const Order = require("../models/Order");

exports.listMine = asyncHandler(async (req, res) => {
  const limit = req.query.limit ? Number(req.query.limit) : null;
  const rows = await Order.findByUser(req.user.id, limit);
  return res.json(new ApiResponse(200, { orders: rows }, "Orders fetched"));
});

exports.getOne = asyncHandler(async (req, res) => {
  const order = await Order.findById(Number(req.params.id));
  if (!order) throw new ApiError(404, "Order not found");
  if (order.userId !== req.user.id && req.user.role !== "admin") {
    throw new ApiError(403, "Not allowed");
  }
  return res.json(new ApiResponse(200, { order }, "Order fetched"));
});

exports.create = asyncHandler(async (req, res) => {
  const { items, subtotal, discount, total, paymentMethod, paymentRef } = req.body;
  if (!Array.isArray(items) || items.length === 0) throw new ApiError(400, "items required");
  const order = await Order.create({
    userId: req.user.id,
    items,
    subtotal: Number(subtotal) || 0,
    discount: Number(discount) || 0,
    total: Number(total) || 0,
    paymentMethod,
    paymentRef,
  });
  return res.status(201).json(new ApiResponse(201, { order }, "Order created"));
});

const db = require("../config/db");

function generateOrderCode() {
  const datePart = new Date().toISOString().slice(2, 10).replace(/-/g, "");
  const rand = Math.floor(1000 + Math.random() * 9000);
  return `EH-${datePart}-${rand}`;
}

const Order = {
  async findByUser(userId, limit = null) {
    const params = [userId];
    let sql = `SELECT o.*,
                      COUNT(oi.id) AS item_count,
                      GROUP_CONCAT(oi.title SEPARATOR ', ') AS items_summary
                 FROM orders o
                 LEFT JOIN order_items oi ON oi.order_id = o.id
                WHERE o.user_id = ?
                GROUP BY o.id
                ORDER BY o.created_at DESC`;
    if (limit) {
      sql += " LIMIT ?";
      params.push(Number(limit));
    }
    const [rows] = await db.query(sql, params);
    return rows.map(Order._row);
  },

  async findById(id) {
    const [rows] = await db.query(
      `SELECT o.*,
              COUNT(oi.id) AS item_count,
              GROUP_CONCAT(oi.title SEPARATOR ', ') AS items_summary
         FROM orders o
         LEFT JOIN order_items oi ON oi.order_id = o.id
        WHERE o.id = ?
        GROUP BY o.id`,
      [id]
    );
    if (!rows.length) return null;
    const order = Order._row(rows[0]);
    const [items] = await db.query(
      "SELECT * FROM order_items WHERE order_id = ?",
      [id]
    );
    order.items = items.map((i) => ({
      id: i.id,
      courseId: i.course_id,
      title: i.title,
      price: Number(i.price),
    }));
    return order;
  },

  /**
   * Create a new order.
   * Accepts an optional `conn` — when provided, the caller controls the transaction.
   */
  async create(
    {
      userId,
      items,
      subtotal,
      discount = 0,
      total,
      status = "pending",
      paymentMethod = null,
      paymentRef = null,
      country = null,
      razorpayOrderId = null,
      razorpayPaymentId = null,
      razorpaySignature = null,
    },
    conn = null
  ) {
    const runner = conn || db;
    const code = generateOrderCode();

    const [res] = await runner.query(
      `INSERT INTO orders
        (user_id, order_code, subtotal, discount, total, status, payment_method, payment_ref,
         country, razorpay_order_id, razorpay_payment_id, razorpay_signature)
       VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      [
        userId,
        code,
        subtotal,
        discount,
        total,
        status,
        paymentMethod,
        paymentRef,
        country,
        razorpayOrderId,
        razorpayPaymentId,
        razorpaySignature,
      ]
    );

    const orderId = res.insertId;

    for (const item of items) {
      await runner.query(
        `INSERT INTO order_items (order_id, course_id, title, price) VALUES (?, ?, ?, ?)`,
        [orderId, item.courseId, item.title, item.price]
      );
    }

    return Order.findById(orderId);
  },

  async attachRazorpayOrderId(orderId, razorpayOrderId) {
    await db.query(`UPDATE orders SET razorpay_order_id = ? WHERE id = ?`, [
      razorpayOrderId,
      orderId,
    ]);
  },

  async findByIdForUser(orderId, userId) {
    const [rows] = await db.query(
      `SELECT * FROM orders WHERE id = ? AND user_id = ? LIMIT 1`,
      [orderId, userId]
    );
    return rows[0] || null;
  },

  /**
   * Mark order paid + return updated row. Must be called inside a transaction.
   */
  async markPaid(
    {
      orderId,
      razorpayPaymentId,
      razorpaySignature,
      paymentMethod = "razorpay",
    },
    conn
  ) {
    const runner = conn || db;
    await runner.query(
      `UPDATE orders
          SET status = 'paid',
              payment_method = ?,
              payment_ref = ?,
              razorpay_payment_id = ?,
              razorpay_signature = ?
        WHERE id = ? AND status = 'pending'`,
      [
        paymentMethod,
        razorpayPaymentId,
        razorpayPaymentId,
        razorpaySignature,
        orderId,
      ]
    );
  },

  _row(r) {
    return {
      id: r.id,
      userId: r.user_id,
      orderCode: r.order_code,
      subtotal: Number(r.subtotal),
      discount: Number(r.discount),
      total: Number(r.total),
      currency: r.currency,
      status: r.status,
      paymentMethod: r.payment_method,
      paymentRef: r.payment_ref,
      country: r.country,
      razorpayOrderId: r.razorpay_order_id,
      razorpayPaymentId: r.razorpay_payment_id,
      createdAt: r.created_at,
      itemCount: Number(r.item_count || 0),
      itemsSummary: r.items_summary || "",
    };
  },
};

module.exports = Order;

const db = require("../config/db");

const Enrollment = {
  async findByUser(userId) {
    const [rows] = await db.query(
      `SELECT e.*,
              c.title AS course_title, c.image_url AS course_image,
              c.course_length, c.category_id,
              cat.name AS category_name
         FROM enrollments e
         JOIN courses c   ON c.id = e.course_id
         LEFT JOIN categories cat ON cat.id = c.category_id
        WHERE e.user_id = ?
        ORDER BY e.last_accessed_at DESC, e.updated_at DESC`,
      [userId]
    );
    return rows.map(Enrollment._row);
  },

  async findOne(userId, courseId) {
    const [rows] = await db.query(
      `SELECT e.*,
              c.title AS course_title, c.image_url AS course_image,
              c.course_length, c.category_id,
              cat.name AS category_name
         FROM enrollments e
         JOIN courses c ON c.id = e.course_id
         LEFT JOIN categories cat ON cat.id = c.category_id
        WHERE e.user_id = ? AND e.course_id = ?`,
      [userId, courseId]
    );
    return rows.length ? Enrollment._row(rows[0]) : null;
  },

  async create({ userId, courseId }) {
    const [res] = await db.query(
      `INSERT INTO enrollments (user_id, course_id, progress, last_accessed_at)
       VALUES (?, ?, 0, NOW())`,
      [userId, courseId]
    );
    return Enrollment.findOne(userId, courseId) || { id: res.insertId };
  },

  async updateProgress(userId, courseId, { progress, lastLesson }) {
    const fields = [];
    const values = [];
    if (progress !== undefined) {
      fields.push("progress = ?");
      values.push(Math.max(0, Math.min(100, Number(progress))));
      if (Number(progress) >= 100) fields.push("completed_at = COALESCE(completed_at, NOW())");
    }
    if (lastLesson !== undefined) {
      fields.push("last_lesson = ?");
      values.push(lastLesson);
    }
    fields.push("last_accessed_at = NOW()");
    values.push(userId, courseId);
    await db.query(
      `UPDATE enrollments SET ${fields.join(", ")} WHERE user_id = ? AND course_id = ?`,
      values
    );
    return Enrollment.findOne(userId, courseId);
  },

  async statsForUser(userId) {
    const [rows] = await db.query(
      `SELECT
         COUNT(*)                                             AS total,
         SUM(CASE WHEN progress >= 100 THEN 1 ELSE 0 END)     AS completed,
         COALESCE(SUM(progress), 0)                           AS progress_sum
       FROM enrollments WHERE user_id = ?`,
      [userId]
    );
    return rows[0] || { total: 0, completed: 0, progress_sum: 0 };
  },

  _row(r) {
    return {
      id: r.id,
      userId: r.user_id,
      courseId: r.course_id,
      progress: r.progress,
      lastLesson: r.last_lesson,
      lastAccessedAt: r.last_accessed_at,
      completedAt: r.completed_at,
      courseTitle: r.course_title,
      courseImage: r.course_image,
      courseLength: r.course_length,
      categoryId: r.category_id,
      categoryName: r.category_name,
    };
  },
};

module.exports = Enrollment;
