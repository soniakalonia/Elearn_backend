const express = require("express");
const cors = require("cors");
const helmet = require("helmet");
const morgan = require("morgan");
const env = require("./config/env");
const routes = require("./routes");
const errorHandler = require("./middlewares/error.middleware");
const ApiError = require("./utils/ApiError");

const app = express();

app.use(helmet());
app.use(
  cors({
    origin: env.FRONTEND_URL,
    credentials: true,
  })
);
app.use(express.json({ limit: "1mb" }));
app.use(morgan("dev"));

app.get("/health", (req, res) => res.json({ ok: true }));
app.use("/api", routes);

app.use((req, res, next) => next(new ApiError(404, "Route not found")));
app.use(errorHandler);

module.exports = app;

CREATE TABLE IF NOT EXISTS `categories` (
  `id`          INT UNSIGNED NOT NULL AUTO_INCREMENT,
  `slug`        VARCHAR(120) NOT NULL,
  `name`        VARCHAR(120) NOT NULL,
  `icon`        VARCHAR(60)  NOT NULL DEFAULT 'grid',
  `blurb`       VARCHAR(255) NOT NULL DEFAULT '',
  `color`       VARCHAR(20)  NOT NULL DEFAULT '#1A1225',
  `soft`        VARCHAR(20)  NOT NULL DEFAULT '#CDF56A',
  `sort_order`  INT          NOT NULL DEFAULT 0,
  `is_active`   TINYINT(1)   NOT NULL DEFAULT 1,
  `created_at`  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at`  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_categories_slug` (`slug`),
  KEY `idx_categories_active` (`is_active`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

const mysql = require("mysql2/promise");
const bcrypt = require("bcryptjs");
const dotenv = require("dotenv");
const path = require("path");

// Load root .env file
dotenv.config({
    path: path.resolve(__dirname, "../../.env")
});

async function seed() {
    let connection;

    try {
        console.log("Starting database seeding...");

        // Validate environment variables
        const requiredEnv = [
            "DB_HOST",
            "DB_USER",
            "DB_PASSWORD",
            "DB_NAME",
            "ADMIN_EMAIL",
            "ADMIN_PASSWORD"
        ];

        for (const key of requiredEnv) {
            if (!process.env[key]) {
                throw new Error(`Missing environment variable: ${key}`);
            }
        }

        // Connect to MySQL
        connection = await mysql.createConnection({
            host: process.env.DB_HOST,
            port: Number(process.env.DB_PORT || 3306),
            user: process.env.DB_USER,
            password: process.env.DB_PASSWORD,
            database: process.env.DB_NAME
        });

        console.log("MySQL connected successfully!");

        // Admin details from .env
        const adminName = process.env.ADMIN_NAME || "Admin";
        const adminEmail = process.env.ADMIN_EMAIL;
        const adminPassword = process.env.ADMIN_PASSWORD;
        const adminPhone = process.env.ADMIN_PHONE || null;

        // Generate secure password hash
        const hashedPassword = await bcrypt.hash(adminPassword, 12);

        // Insert admin or update existing admin
        const query = `
            INSERT INTO users
            (
                name,
                email,
                phone,
                password,
                role,
                is_verified,
                is_active
            )
            VALUES (?, ?, ?, ?, 'admin', 1, 1)

            ON DUPLICATE KEY UPDATE
                name = VALUES(name),
                phone = VALUES(phone),
                password = VALUES(password),
                role = 'admin',
                is_verified = 1,
                is_active = 1
        `;

        await connection.execute(query, [
            adminName,
            adminEmail,
            adminPhone,
            hashedPassword
        ]);

                // Seed categories
        const categories = [
          { slug: "devops-cloud", name: "DevOps & Cloud", icon: "cloud", blurb: "Ship faster, break less.", color: "#1A1225", soft: "#CDF56A" },
          { slug: "cad-design", name: "CAD & Design", icon: "compass", blurb: "Draft, model, build.", color: "#241533", soft: "#D9C8FF" },
          { slug: "data-ai", name: "Data & AI", icon: "sparkle", blurb: "Turn data into decisions.", color: "#1A1225", soft: "#A9DDF5" },
          { slug: "cybersecurity", name: "Cybersecurity", icon: "shield", blurb: "Defend what matters.", color: "#1A1225", soft: "#FFD3C7" },
          { slug: "web-development", name: "Web Development", icon: "code", blurb: "Build for the browser.", color: "#1A1225", soft: "#B9ECD4" },
          { slug: "ux-ui", name: "Product & UX", icon: "pen", blurb: "Design products people love.", color: "#241533", soft: "#FFE3A8" },
          { slug: "business-analytics", name: "Business & Analytics", icon: "chart", blurb: "Make numbers make sense.", color: "#1A1225", soft: "#F2D6FF" },
          { slug: "ai-fluency", name: "AI Fluency", icon: "bot", blurb: "Work smarter with AI.", color: "#1A1225", soft: "#CDF56A" },
        ];

        const catQuery = `
          INSERT INTO categories (slug, name, icon, blurb, color, soft, sort_order, is_active)
          VALUES (?, ?, ?, ?, ?, ?, ?, 1)
          ON DUPLICATE KEY UPDATE
            name = VALUES(name),
            icon = VALUES(icon),
            blurb = VALUES(blurb),
            color = VALUES(color),
            soft = VALUES(soft),
            sort_order = VALUES(sort_order)
        `;

        for (let i = 0; i < categories.length; i++) {
          const c = categories[i];
          await connection.execute(catQuery, [c.slug, c.name, c.icon, c.blurb, c.color, c.soft, i]);
        }

        console.log(`${categories.length} categories seeded`);

        
        console.log("ADMIN SEEDED SUCCESSFULLY!");
      
        console.log("Admin Name:", adminName);
        console.log("Admin Email:", adminEmail);
        console.log("Admin Role: admin");
        console.log("Password: Securely hashed");

    } catch (error) {
        console.error("Database seeding failed:", error.message);
        process.exitCode = 1;

    } finally {
        if (connection) {
            await connection.end();
        }
    }
}

seed();
