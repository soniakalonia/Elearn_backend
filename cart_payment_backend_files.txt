const pool = require("../config/db");
const ApiError = require("../utils/ApiError");

async function listCart(userId) {
  const [rows] = await pool.execute(
    `SELECT ci.id, ci.course_id AS courseId, c.title, c.discount_price AS price, c.image_url AS imageUrl
     FROM cart_items ci
     JOIN courses c ON c.id = ci.course_id
     WHERE ci.user_id = ?
     ORDER BY ci.created_at DESC`,
    [userId]
  );
  return rows.map((r) => ({ ...r, price: Number(r.price) }));
}

async function addToCart(userId, courseId) {
  const [course] = await pool.execute("SELECT id FROM courses WHERE id = ? LIMIT 1", [courseId]);
  if (!course[0]) throw new ApiError(404, "Course not found");

  await pool.execute(
    "INSERT IGNORE INTO cart_items (user_id, course_id) VALUES (?, ?)",
    [userId, courseId]
  );
  return listCart(userId);
}

async function removeFromCart(userId, courseId) {
  await pool.execute(
    "DELETE FROM cart_items WHERE user_id = ? AND course_id = ?",
    [userId, courseId]
  );
  return listCart(userId);
}

module.exports = { listCart, addToCart, removeFromCart };

const db = require("../config/db");

const Order = {
  async findByUser(userId, limit = null) {
    const params = [userId];
    let sql = `SELECT o.*,
                      COUNT(oi.id) AS item_count,
                      GROUP_CONCAT(oi.title SEPARATOR ', ') AS items_summary
                 FROM orders o
                 LEFT JOIN order_items oi ON oi.order_id = o.id
                WHERE o.user_id = ?
                GROUP BY o.id
                ORDER BY o.created_at DESC`;
    if (limit) {
      sql += " LIMIT ?";
      params.push(Number(limit));
    }
    const [rows] = await db.query(sql, params);
    return rows.map(Order._row);
  },

  async findById(id) {
    const [rows] = await db.query(
      `SELECT o.*,
              COUNT(oi.id) AS item_count,
              GROUP_CONCAT(oi.title SEPARATOR ', ') AS items_summary
         FROM orders o
         LEFT JOIN order_items oi ON oi.order_id = o.id
        WHERE o.id = ?
        GROUP BY o.id`,
      [id]
    );
    if (!rows.length) return null;
    const order = Order._row(rows[0]);
    const [items] = await db.query("SELECT * FROM order_items WHERE order_id = ?", [id]);
    order.items = items.map((i) => ({ id: i.id, courseId: i.course_id, title: i.title, price: Number(i.price) }));
    return order;
  },

  async create({ userId, items, subtotal, discount, total, paymentMethod, paymentRef }) {
    const code = `EH-${new Date().toISOString().slice(2, 10).replace(/-/g, "")}-${Math.floor(1000 + Math.random() * 9000)}`;
    const [res] = await db.query(
      `INSERT INTO orders (user_id, order_code, subtotal, discount, total, status, payment_method, payment_ref)
       VALUES (?, ?, ?, ?, ?, 'paid', ?, ?)`,
      [userId, code, subtotal, discount, total, paymentMethod || "manual", paymentRef || null]
    );
    const orderId = res.insertId;
    for (const item of items) {
      await db.query(
        `INSERT INTO order_items (order_id, course_id, title, price) VALUES (?, ?, ?, ?)`,
        [orderId, item.courseId, item.title, item.price]
      );
    }
    return Order.findById(orderId);
  },

  _row(r) {
    return {
      id: r.id,
      userId: r.user_id,
      orderCode: r.order_code,
      subtotal: Number(r.subtotal),
      discount: Number(r.discount),
      total: Number(r.total),
      currency: r.currency,
      status: r.status,
      paymentMethod: r.payment_method,
      paymentRef: r.payment_ref,
      createdAt: r.created_at,
      itemCount: Number(r.item_count || 0),
      itemsSummary: r.items_summary || "",
    };
  },
};

module.exports = Order;

const pool = require("../config/db");

const User = {
  async countActive() {
    const [rows] = await pool.execute(
      "SELECT COUNT(*) AS n FROM users WHERE role = 'user' AND is_active = 1"
    );
    return Number(rows[0].n);
  },

  async countNewByDay(days = 30) {
    const [rows] = await pool.execute(
      `SELECT DATE(created_at) AS d, COUNT(*) AS n
         FROM users
        WHERE role = 'user' AND created_at >= DATE_SUB(CURDATE(), INTERVAL ? DAY)
        GROUP BY DATE(created_at)
        ORDER BY d ASC`,
      [days]
    );
    return rows.map((r) => ({ date: String(r.d), count: Number(r.n) }));
  },

  async countAll() {
    const [rows] = await pool.execute("SELECT COUNT(*) AS n FROM users WHERE role = 'user'");
    return Number(rows[0].n);
  },
};

module.exports = User;

const pool = require("../config/db");
const ApiError = require("../utils/ApiError");

const publicUser = (u) => ({
  id: u.id,
  name: u.name,
  email: u.email,
  phone: u.phone,
  role: u.role,
  avatar: u.avatar,
  isVerified: !!u.is_verified,
  isActive: !!u.is_active,
  lastLoginAt: u.last_login_at,
  createdAt: u.created_at,
});

async function listUsers({ page = 1, limit = 20, search = "", role }) {
  const offset = (page - 1) * limit;
  const where = [];
  const params = [];

  if (search) {
    where.push("(name LIKE ? OR email LIKE ?)");
    const like = `%${search}%`;
    params.push(like, like);
  }
  if (role === "user" || role === "admin") {
    where.push("role = ?");
    params.push(role);
  }

  const whereSql = where.length ? `WHERE ${where.join(" AND ")}` : "";

  const [countRows] = await pool.execute(
    `SELECT COUNT(*) AS total FROM users ${whereSql}`,
    params
  );

  const [rows] = await pool.execute(
    `SELECT id, name, email, phone, role, avatar, is_verified, is_active, last_login_at, created_at
     FROM users ${whereSql}
     ORDER BY created_at DESC
     LIMIT ? OFFSET ?`,
    [...params, Number(limit), Number(offset)]
  );

  return {
    users: rows.map(publicUser),
    total: countRows[0].total,
    page: Number(page),
    limit: Number(limit),
  };
}

async function setActive(userId, isActive) {
  const [result] = await pool.execute(
    "UPDATE users SET is_active = ? WHERE id = ?",
    [isActive ? 1 : 0, userId]
  );
  if (!result.affectedRows) throw new ApiError(404, "User not found");

  const [rows] = await pool.execute(
    "SELECT id, name, email, phone, role, avatar, is_verified, is_active, last_login_at, created_at FROM users WHERE id = ?",
    [userId]
  );
  return publicUser(rows[0]);
}

module.exports = { listUsers, setActive };

const ApiError = require("../utils/ApiError");
const { verifyToken } = require("../utils/jwt");
const asyncHandler = require("../utils/asyncHandler");
const pool = require("../config/db");

const auth = asyncHandler(async (req, res, next) => {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : null;
  if (!token) throw new ApiError(401, "Not authenticated");

  let decoded;
  try {
    decoded = verifyToken(token);
  } catch {
    throw new ApiError(401, "Invalid or expired token");
  }

  const [rows] = await pool.execute(
    "SELECT id, name, email, phone, role, avatar, is_verified, is_active FROM users WHERE id = ? LIMIT 1",
    [decoded.id]
  );
  const user = rows[0];
  if (!user) throw new ApiError(401, "User no longer exists");
  if (!user.is_active) throw new ApiError(403, "Account is disabled");

  req.user = user;
  next();
});

module.exports = auth;

const ApiError = require("../utils/ApiError");

const validate = (schema) => (req, res, next) => {
  const { error, value } = schema.validate(req.body, {
    abortEarly: false,
    stripUnknown: true,
  });
  if (error) {
    return next(
      new ApiError(400, error.details.map((d) => d.message).join(", "))
    );
  }
  req.body = value;
  next();
};

module.exports = validate;

class ApiError extends Error {
  constructor(statusCode, message) {
    super(message);
    this.statusCode = statusCode;
    this.isOperational = true;
    Error.captureStackTrace(this, this.constructor);
  }
}

module.exports = ApiError;
